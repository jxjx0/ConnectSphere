import React, { useEffect, useState } from 'react'
import { useParams } from 'react-router-dom'
import { getSafetyEventDetails } from '../api/events'

export default function SafetyReviewDetailPage() {
  const { eventId } = useParams()
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    const token = localStorage.getItem('connectsphere.token')
    getSafetyEventDetails(eventId, token)
      .then((res) => setData(res))
      .catch((err) => setError(err.message))
      .finally(() => setLoading(false))
  }, [eventId])

  if (loading) return <div className="p-6 text-stone-600">Loading safety assessment details...</div>
  if (error) return <div className="p-6 text-red-600 font-medium">Error: {error}</div>

  const { event, venue_booking, venue, equipment_items, safety_flags } = data

  return (
    <div className="p-8 max-w-5xl mx-auto space-y-6">
      <div className="flex items-center justify-between border-b pb-4">
        <div>
          <h1 className="text-2xl font-bold text-stone-900">{event.name}</h1>
          <p className="text-sm text-stone-500">
            Category: <span className="capitalize">{event.category}</span>
          </p>
        </div>
        <span className="px-3 py-1 text-xs rounded-full bg-orange-100 text-orange-800 border border-orange-300 font-medium">
          Safety check not done
        </span>
      </div>

      {/* Safety Flags / Clashes Warning Box */}
      {safety_flags && safety_flags.length > 0 && (
        <div className="bg-red-50 border border-red-200 p-4 rounded-lg space-y-2">
          <h3 className="text-red-800 font-semibold flex items-center gap-2">
            ⚠️ Safety Concerns / Clashes Detected
          </h3>
          {safety_flags.map((flag, idx) => (
            <p key={idx} className="text-sm text-red-700">
              <strong>[{flag.type}]</strong> {flag.message}
            </p>
          ))}
        </div>
      )}

      {/* Event Details Section */}
      <div className="bg-white border border-stone-200 rounded-lg p-6 space-y-3 shadow-sm">
        <h2 className="text-lg font-semibold text-stone-800 border-b pb-2">Event Overview</h2>
        <div className="grid grid-cols-2 gap-4 text-sm">
          <p><strong>Description:</strong> {event.description || 'N/A'}</p>
          <p><strong>Purpose:</strong> {event.purpose || 'N/A'}</p>
          <p><strong>Date & Time:</strong> {new Date(event.start_time).toLocaleString()} – {new Date(event.end_time).toLocaleString()}</p>
          <p><strong>Expected Attendance:</strong> {event.expected_attendance || 'N/A'}</p>
          <p><strong>Coordinator in Charge:</strong> {event.coordinator?.full_name || 'Unassigned'}</p>
          <p><strong>Contact Number:</strong> {event.coordinator?.contact_number || 'N/A'}</p>
        </div>
      </div>

      {/* Venue Profile, Layout & Accessibility Section */}
      <div className="bg-white border border-stone-200 rounded-lg p-6 space-y-3 shadow-sm">
        <h2 className="text-lg font-semibold text-stone-800 border-b pb-2">Venue Profile & Layout</h2>
        <div className="grid grid-cols-2 gap-4 text-sm">
          <p><strong>Venue Name:</strong> {venue.name || 'No Venue Assigned'}</p>
          <p><strong>Location:</strong> {venue.location || 'N/A'}</p>
          <p><strong>Capacity Limit:</strong> {venue.capacity || 'N/A'}</p>
          <p><strong>Requested Room Layout:</strong> {venue_booking.room_layout || event.layout_preference || 'N/A'}</p>
        </div>

        <div className="pt-2 text-sm space-y-2">
          <p>
            <strong>Accessibility Features:</strong>{' '}
            {venue.accessibility ? JSON.stringify(venue.accessibility) : event.accessibility_requirements || 'None'}
          </p>
          <p>
            <strong>Facilities Available:</strong>{' '}
            {venue.facilities ? JSON.stringify(venue.facilities) : 'N/A'}
          </p>
          <p>
            <strong>Operating Hours:</strong>{' '}

// Roles arrive as 'venue_staff'; people read "Venue staff".
function readableRole(role) {
  const words = role.replace(/_/g, ' ')
  return words.charAt(0).toUpperCase() + words.slice(1)
}

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:5000'

async function request(path, { method = 'GET', body, token } = {}) {
  const headers = {}
  if (body) headers['Content-Type'] = 'application/json'
  if (token) headers.Authorization = `Bearer ${token}`

  const response = await fetch(`${API_BASE_URL}${path}`, {
    method,
    headers,
    body: body ? JSON.stringify(body) : undefined,
  })

  const data = await response.json().catch(() => ({}))

  if (!response.ok) {
    const message = data.errors ? data.errors.join(' ') : data.error
    const error = new Error(message || 'Something went wrong. Please try again.')
    // Lets a page react to a specific status, e.g. 409 when someone else
    // has already changed the record.
    error.status = response.status
    throw error
  }

  return data
}
